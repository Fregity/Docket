import type { MetadataRoute } from "next";

export default function robots(): MetadataRoute.Robots {
  return {
    rules: {
      userAgent: "*",
      allow: "/",
      disallow: [
        "/admin",
        "/auth/",
        "/classes",
        "/dashboard",
        "/history",
        "/onboarding",
        "/settings",
      ],
    },
    sitemap: "https://myduebird.com/sitemap.xml",
  };
}
